        }
        System.out.println();
    }
    public int[] reverseArray(int[] arr){
        int i = 0;
        int j = arr.length - 1;
        while(i<j){
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            i++;
            j--;
        }
        return arr;
    }
}

// Check Prime Number or not

import java.util.Scanner;

class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        System.out.println("Enter Number :- ");
        int num = sc.nextInt();
        Main obj = new Main();
        System.out.println(obj.isPrime(num));
    }
    public boolean isPrime(int num){
        if(num<=1){
            return false;
        }
        for(int i=2;i<num;i++){
            if(num%2==0){
                return false;
            }
        }
        return true;
    }
}

// }}}how many times number here in array find the countes of same number present in array 


// const arr = [12,23,34,45,56,67,12,234,45,56,78,12,34,89,90,45,90,56,90,34,2,90];


import java.util.HashMap;

class Main{
    public static void main(String[] args){
        int[] arr = {
            12,23,34,45,56,67,12,234,45,56,78,12,34,89,90,45,90,56,90,90,34,2,90
        };
        Main obj = new Main();
        obj.countesNumber(arr);
    }
    public void countesNumber(int[] arr){
        HashMap<Integer,Integer> counts = new HashMap<>();
        for(int i=0;i<arr.length;i++){
            int num = arr[i];
            if(counts.containsKey(num)){
                counts.put(num,counts.get(num)+1);
            }else{
                counts.put(num,1);
            }
        }
        System.out.println(counts);
    }
}

// }}}bubble sort 1D,2D
// const arr = [[12, 23, 90, 78, 4, 89], [45, 99, 1, 8, 87], [21, 32, 2]];


class Main{
    public static void main(String[] args){
        int arr[][] = {
            {12, 23, 90, 78, 4, 89},
            {45, 99, 1, 8, 87},
            {21, 32, 2}
        };
        int [][] newSortArray;
        Main obj = new Main();
        newSortArray = obj.twoDBubbleSort(arr);
        printArray(newSortArray);
    }
    public int[][] twoDBubbleSort(int[][] arr){
        // int newArray[][];
        for(int i=0;i<arr.length;i++){
            arr[i] = sortArray(arr[i]);
        }
        return arr;
    }
    public static int[] sortArray(int[] arr){
        for(int i=0;i<arr.length;i++){
            for(int j=i+1;j<arr.length;j++){
                if(arr[i]>arr[j]){
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = temp;
                }
            }
        }
        return arr;
    }
    public static void printArray(int[][] arr){
        for(int i=0;i<arr.length;i++){
            for(int j=0;j<arr[i].length;j++){
                System.out.print(arr[i][j]+" ");
            }
            System.out.println();
        }
    }
}

// }}}find maximum number and smallest number in array 
// const arr = [23,45,89,90,12,78,44];

class Main
{
    public static void main(String[] args){
        int arr[] = {23,45,89,90,12,78,44};
        Main obj = new Main();
        System.out.println("Maximum Number :- "+obj.findMaxNumber(arr));
    }
    public int findMaxNumber(int[] arr){
        int max = arr[0];
        for(int i=1;i<arr.length;i++){
            if(arr[i]>max){
                max = arr[i];
            }
        }
        return max;
    }
}
